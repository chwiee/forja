exemplo.go
package exemplo

import (
	"context"
	"errors"
	"fmt"
)

// Erro "sentinela": o chamador compara com errors.Is.
var ErrNotFound = errors.New("não encontrado")

// Interface: descreve o que eu preciso, não quem implementa.
type Store interface {
	Get(ctx context.Context, key string) (string, error)
}

// Struct com dependência injetada (a "classe").
type Service struct {
	store Store // minúsculo: privado ao pacote
}

// Construtor: uma função que devolve o struct pronto.
func NewService(s Store) *Service { return &Service{store: s} }

// Método com receiver ponteiro. Maiúsculo: exportado.
func (s *Service) Describe(ctx context.Context, key string) (string, error) {
	v, err := s.store.Get(ctx, key)
	if errors.Is(err, ErrNotFound) {
		return "(vazio)", nil
	}
	if err != nil {
		return "", fmt.Errorf("lendo %q: %w", key, err) // %w mantém errors.Is/As
	}
	return v, nil
}
